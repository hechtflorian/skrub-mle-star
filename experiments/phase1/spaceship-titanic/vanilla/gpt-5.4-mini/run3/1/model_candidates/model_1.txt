## Model name
CatBoostClassifier

## Example Python code
import pandas as pd
from catboost import CatBoostClassifier
from sklearn.model_selection import train_test_split

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

def prep(df):
    df = df.copy()
    df['Deck'] = df['Cabin'].astype(str).str.split('/').str[0]
    df['Side'] = df['Cabin'].astype(str).str.split('/').str[2]
    df['Group'] = df['PassengerId'].astype(str).str.split('_').str[0]
    df['Spending'] = df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']].sum(axis=1)
    return df.drop(columns=['Cabin','Name'])

train_p = prep(train)
test_p = prep(test)
X = train_p.drop(columns=['Transported'])
y = train_p['Transported'].astype(int)
cat = ['HomePlanet','CryoSleep','Destination','VIP','Deck','Side','Group']

model = CatBoostClassifier(loss_function='Logloss', depth=8, learning_rate=0.05, iterations=2000, verbose=0)
model.fit(X, y, cat_features=cat)
pred = model.predict(test_p)
out = pd.DataFrame({'PassengerId': test['PassengerId'], 'Transported': pred.astype(bool).ravel()})
out.to_csv('submission.csv', index=False)
