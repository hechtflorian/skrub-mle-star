## Model name
Tuned HistGradientBoostingRegressor with DataOps hyperparameter choices

## Example Python code
import pandas as pd
import skrub
from sklearn.ensemble import HistGradientBoostingRegressor

train = pd.read_csv("train.csv")
test = pd.read_csv("test.csv")

X = skrub.X(train.drop(columns=["median_house_value"]))
y = skrub.y(train["median_house_value"])

model = HistGradientBoostingRegressor(
    learning_rate=skrub.choose_float(0.01, 0.2, log=True, name="lr"),
    max_depth=skrub.choose_int(3, 10, name="max_depth"),
    max_leaf_nodes=skrub.choose_int(15, 63, name="max_leaf_nodes"),
    min_samples_leaf=skrub.choose_int(10, 50, name="min_samples_leaf"),
    l2_regularization=skrub.choose_float(1e-6, 1.0, log=True, name="l2"),
    random_state=0,
)

pred = X.skb.apply(model, y=y)
search = pred.skb.make_randomized_search(
    scoring="neg_root_mean_squared_error",
    n_iter=24,
    random_state=0,
    fitted=True,
)

best = search.best_learner_
submission = pd.DataFrame({"median_house_value": best.predict(test)})
submission.to_csv("submission.csv", index=False)
