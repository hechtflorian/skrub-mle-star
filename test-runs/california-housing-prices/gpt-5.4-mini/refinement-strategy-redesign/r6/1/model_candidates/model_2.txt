## Example Python code
```python
import pandas as pd
import lightgbm as lgb

X_train = train_df.drop(columns=['median_house_value'])
y_train = train_df['median_house_value']
X_test = test_df.copy()

model = lgb.LGBMRegressor(
    n_estimators=5000,
    learning_rate=0.03,
    num_leaves=64,
    subsample=0.8,
    colsample_bytree=0.8,
    random_state=42,
)
model.fit(
    X_train, y_train,
    eval_set=[(X_train, y_train)],
    eval_metric='rmse',
    callbacks=[lgb.early_stopping(200, verbose=False)]
)

pred = model.predict(X_test)
pd.DataFrame({'median_house_value': pred}).to_csv('submission.csv', index=False)
```
