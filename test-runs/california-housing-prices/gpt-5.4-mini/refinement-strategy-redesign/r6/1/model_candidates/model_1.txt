## Example Python code
```python
import pandas as pd
from catboost import CatBoostRegressor, Pool

X_train = train_df.drop(columns=['median_house_value'])
y_train = train_df['median_house_value']
X_test = test_df.copy()

model = CatBoostRegressor(
    loss_function='RMSE',
    depth=8,
    learning_rate=0.05,
    iterations=3000,
    random_seed=42,
    verbose=200,
)
model.fit(X_train, y_train, eval_set=Pool(X_train, y_train), use_best_model=True)

pred = model.predict(X_test)
pd.DataFrame({'median_house_value': pred}).to_csv('submission.csv', index=False)
```
