## Model name
CatBoostRegressor + skrub.TableVectorizer

## Example Python code
import pandas as pd
import skrub
from catboost import CatBoostRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error

train_df = pd.read_csv('train.csv')
test_df = pd.read_csv('test.csv')
target = 'median_house_value'

train_df[target] = train_df[target].fillna(train_df[target].median())
X_train, X_val, y_train, y_val = train_test_split(
    train_df.drop(columns=[target]), train_df[target], test_size=0.2, random_state=42
)

pipe = skrub.X(X_train).skb.apply(
    skrub.TableVectorizer()
).skb.apply(
    CatBoostRegressor(loss_function='RMSE', depth=8, learning_rate=0.05, iterations=2000, verbose=False),
    y=skrub.y(y_train)
)

learner = pipe.skb.make_learner(fitted=True)
val_pred = learner.predict({'data': X_val})
rmse = mean_squared_error(y_val, val_pred) ** 0.5
print(f'Final Validation Performance: {rmse}')

test_pred = learner.predict({'data': test_df})
pd.DataFrame({target: test_pred}).to_csv('submission.csv', index=False)
