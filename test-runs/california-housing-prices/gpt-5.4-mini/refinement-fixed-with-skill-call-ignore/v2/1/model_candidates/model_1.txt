## Model name
CatBoostRegressor with skrub preprocessing (DataOps-friendly)

## Example Python code
import pandas as pd
import skrub
from catboost import CatBoostRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')
target = 'median_house_value'

X = skrub.X(train.drop(columns=[target]))
y = skrub.y(train[target])

pipe = X.skb.apply(skrub.TableVectorizer()).skb.apply(
    CatBoostRegressor(
        loss_function='RMSE',
        iterations=2000,
        depth=8,
        learning_rate=0.03,
        random_seed=0,
        verbose=0,
    ),
    y=y,
)

learner = pipe.skb.make_learner(fitted=True)
preds = learner.predict({'data': test})

pd.DataFrame({target: preds}).to_csv('submission.csv', index=False)
