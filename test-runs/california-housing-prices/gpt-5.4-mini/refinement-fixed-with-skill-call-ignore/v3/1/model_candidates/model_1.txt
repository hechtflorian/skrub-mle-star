## Model name
Skrub tabular_pipeline + HistGradientBoostingRegressor

## Example Python code
import pandas as pd
import skrub
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')
target = 'median_house_value'

X_train, X_valid, y_train, y_valid = train_test_split(
    train.drop(columns=[target]), train[target], test_size=0.2, random_state=42
)

pipe = skrub.tabular_pipeline('regression')
pipe.fit(X_train, y_train)
pred = pipe.predict(X_valid)
rmse = mean_squared_error(y_valid, pred) ** 0.5
print(f'Final Validation Performance: {rmse}')

test_pred = pipe.predict(test)
pd.DataFrame({'median_house_value': test_pred}).to_csv('submission.csv', index=False)
