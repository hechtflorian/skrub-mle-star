## Example Python code
import pandas as pd
import skrub
from sklearn.model_selection import cross_val_score

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

y = train['median_house_value']
X = train.drop(columns=['median_house_value'])

# strong default for tabular regression in skrub
pipe = skrub.tabular_pipeline('regression', random_state=42)

# optional CV check
scores = cross_val_score(pipe, X, y, cv=5, scoring='neg_root_mean_squared_error')
print(-scores.mean())

pipe.fit(X, y)
preds = pipe.predict(test)
pd.DataFrame({'median_house_value': preds}).to_csv('submission.csv', index=False)
