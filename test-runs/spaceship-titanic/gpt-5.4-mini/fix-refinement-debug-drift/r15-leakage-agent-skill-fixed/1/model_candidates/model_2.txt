## Model name
LightGBM LGBMClassifier

## Example Python code
import pandas as pd
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import accuracy_score
from lightgbm import LGBMClassifier

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')
all_df = pd.concat([train.drop(columns='Transported'), test], axis=0)

all_df[['Deck','Num','Side']] = all_df['Cabin'].str.split('/', expand=True)
all_df[['First','Last']] = all_df['Name'].str.split(' ', 1, expand=True)
for c in ['HomePlanet','CryoSleep','Destination','VIP','Deck','Side']:
    all_df[c] = all_df[c].fillna('Missing').astype('category')
for c in ['Age','RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']:
    all_df[c] = all_df[c].fillna(0)
all_df = all_df.drop(['PassengerId','Cabin','Name'], axis=1)

X = all_df.iloc[:len(train)]
y = train['Transported'].astype(int)
X_test = all_df.iloc[len(train):]

model = LGBMClassifier(
    n_estimators=3000,
    learning_rate=0.03,
    num_leaves=31,
    subsample=0.8,
    colsample_bytree=0.8,
    random_state=42
)
model.fit(X, y)
preds = model.predict(X_test).astype(bool)
sub = pd.DataFrame({'PassengerId': test['PassengerId'], 'Transported': preds})
sub.to_csv('submission.csv', index=False)
