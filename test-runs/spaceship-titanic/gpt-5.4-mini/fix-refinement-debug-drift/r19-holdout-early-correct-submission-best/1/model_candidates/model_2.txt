LightGBM (LGBMClassifier)

## Example Python code
import pandas as pd
from lightgbm import LGBMClassifier

for df in [train_df, test_df]:
    cabin = df['Cabin'].fillna('Missing/Missing/Missing').str.split('/', expand=True)
    df['Deck'] = cabin[0]
    df['Side'] = cabin[2]
    df['TotalSpent'] = df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']].fillna(0).sum(axis=1)

train = pd.get_dummies(train_df.drop(columns=['Transported']), dummy_na=True)
test = pd.get_dummies(test_df, dummy_na=True)
train, test = train.align(test, join='left', axis=1, fill_value=0)

y = train_df['Transported'].astype(int)
model = LGBMClassifier(
    n_estimators=2000,
    learning_rate=0.03,
    num_leaves=64,
    subsample=0.8,
    colsample_bytree=0.8,
    random_state=42,
    verbose=-1
)
model.fit(train, y)
pred = model.predict(test).astype(bool)
submission = pd.DataFrame({'PassengerId': test_df['PassengerId'], 'Transported': pred})
submission.to_csv('submission.csv', index=False)
