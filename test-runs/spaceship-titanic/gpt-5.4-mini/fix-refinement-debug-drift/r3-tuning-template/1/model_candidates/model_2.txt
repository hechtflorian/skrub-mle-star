## Model name
LightGBM (LGBMClassifier)

## Example Python code
import pandas as pd
from sklearn.model_selection import train_test_split
from lightgbm import LGBMClassifier

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

# simple feature engineering
for df in [train, test]:
    df[['Deck','Num','Side']] = df['Cabin'].str.split('/', expand=True)
    df['Group'] = df['PassengerId'].str.split('_').str[0]
    df['Spending'] = df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']].sum(axis=1)
    df.drop(columns=['Name','Cabin'], inplace=True)

X = pd.get_dummies(train.drop(columns=['Transported']), dummy_na=True)
y = train['Transported'].astype(int)
X_test = pd.get_dummies(test, dummy_na=True)
X_test = X_test.reindex(columns=X.columns, fill_value=0)

model = LGBMClassifier(
    n_estimators=1200,
    learning_rate=0.03,
    num_leaves=31,
    subsample=0.8,
    colsample_bytree=0.8,
    random_state=42,
    n_jobs=-1,
    verbose=-1
)
model.fit(X, y)
preds = model.predict(X_test).astype(bool)
sub = pd.DataFrame({'PassengerId': test['PassengerId'], 'Transported': preds})
sub.to_csv('submission.csv', index=False)
