## Model name
CatBoostClassifier

## Example Python code
import pandas as pd
from catboost import CatBoostClassifier, Pool

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

for df in [train, test]:
    cabin = df['Cabin'].fillna('Unknown/0/U').str.split('/', expand=True)
    df['Deck'] = cabin[0]
    df['Num'] = pd.to_numeric(cabin[1], errors='coerce')
    df['Side'] = cabin[2]
    df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']] = df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']].fillna(0)
    df['TotalSpending'] = df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']].sum(axis=1)
    df['NameLen'] = df['Name'].fillna('').str.len()

X = train.drop(columns=['Transported'])
y = train['Transported'].astype(int)
cat_cols = ['HomePlanet','CryoSleep','VIP','Deck','Side','Destination']

model = CatBoostClassifier(
    iterations=2000, learning_rate=0.03, depth=8,
    loss_function='Logloss', eval_metric='Accuracy',
    random_seed=42, verbose=0
)
model.fit(X, y, cat_features=cat_cols)
pred = model.predict(test)
sub = pd.DataFrame({'PassengerId': test['PassengerId'], 'Transported': pred.astype(bool).ravel()})
sub.to_csv('submission.csv', index=False)
