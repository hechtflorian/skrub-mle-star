## Model name
LightGBMClassifier

## Example Python code
import pandas as pd
from lightgbm import LGBMClassifier

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')
all_df = pd.concat([train.drop(columns='Transported'), test], axis=0)

for df in [all_df]:
    cabin = df['Cabin'].fillna('Unknown/0/U').str.split('/', expand=True)
    df['Deck'] = cabin[0]
    df['Num'] = pd.to_numeric(cabin[1], errors='coerce')
    df['Side'] = cabin[2]
    df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']] = df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']].fillna(0)
    df['TotalSpending'] = df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']].sum(axis=1)
    df['GroupSize'] = df['PassengerId'].str.split('_').str[0].map(df['PassengerId'].str.split('_').str[0].value_counts())
    df['NameLen'] = df['Name'].fillna('').str.len()

train_X = all_df.iloc[:len(train)].copy()
test_X = all_df.iloc[len(train):].copy()

y = train['Transported'].astype(int)
cat_cols = ['HomePlanet','CryoSleep','VIP','Deck','Side','Destination']
for c in cat_cols:
    train_X[c] = train_X[c].astype('category')
    test_X[c] = test_X[c].astype('category')

model = LGBMClassifier(
    n_estimators=2000, learning_rate=0.03, num_leaves=31,
    subsample=0.8, colsample_bytree=0.8, random_state=42,
    n_jobs=-1
)
model.fit(train_X.drop(columns=['PassengerId']), y, categorical_feature=cat_cols)
pred = model.predict(test_X.drop(columns=['PassengerId']))
sub = pd.DataFrame({'PassengerId': test['PassengerId'], 'Transported': pred.astype(bool)})
sub.to_csv('submission.csv', index=False)
