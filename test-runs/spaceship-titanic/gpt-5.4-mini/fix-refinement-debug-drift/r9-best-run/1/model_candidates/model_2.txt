import pandas as pd
from lightgbm import LGBMClassifier
from sklearn.model_selection import train_test_split
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder

for df in [train_df, test_df]:
    df['CabinDeck'] = df['Cabin'].fillna('X').astype(str).str.split('/').str[0]
    df['CabinSide'] = df['Cabin'].fillna('X').astype(str).str.split('/').str[-1]
    df['TotalSpend'] = df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']].fillna(0).sum(axis=1)

features = ['HomePlanet','CryoSleep','CabinDeck','CabinSide','Destination','Age','VIP','TotalSpend']
X = train_df[features]
y = train_df['Transported'].astype(int)
Xt = test_df[features]

pre = ColumnTransformer([
    ('cat', OneHotEncoder(handle_unknown='ignore'), ['HomePlanet','CryoSleep','CabinDeck','CabinSide','Destination','VIP'])
], remainder='passthrough')

model = Pipeline([
    ('pre', pre),
    ('clf', LGBMClassifier(
        n_estimators=800,
        learning_rate=0.03,
        num_leaves=31,
        subsample=0.9,
        colsample_bytree=0.8,
        random_state=42,
        verbose=-1
    ))
])
model.fit(X, y)
pred = model.predict(Xt).astype(bool)
submission = pd.DataFrame({'PassengerId': test_df['PassengerId'], 'Transported': pred})
submission.to_csv('submission.csv', index=False)
