## Model name
CatBoostRegressor

## Example Python code
import pandas as pd
from catboost import CatBoostRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')
X = train.drop(columns=['id', 'yield'])
y = train['yield']

model = CatBoostRegressor(
    depth=8,
    learning_rate=0.05,
    iterations=5000,
    loss_function='MAE',
    random_seed=42,
    verbose=0
)
model.fit(X, y)
pred = model.predict(test.drop(columns=['id']))
sub = pd.DataFrame({'id': test['id'], 'yield': pred})
sub.to_csv('submission.csv', index=False)
