## Example Python code
import pandas as pd
from catboost import CatBoostRegressor

# simple time features
for df in [train_df, test_df]:
    df['Date'] = pd.to_datetime(df['Date'])
    df['day'] = df['Date'].dt.day
    df['month'] = df['Date'].dt.month
    df['dow'] = df['Date'].dt.dayofweek
    df['doy'] = df['Date'].dt.dayofyear
    df['country_province'] = df['Country_Region'].fillna('') + '_' + df['Province_State'].fillna('')

feat = ['Country_Region', 'Province_State', 'day', 'month', 'dow', 'doy', 'country_province']
X_train, X_test = train_df[feat], test_df[feat]

m1 = CatBoostRegressor(loss_function='RMSE', verbose=0, random_seed=42)
m2 = CatBoostRegressor(loss_function='RMSE', verbose=0, random_seed=42)
m1.fit(X_train, train_df['ConfirmedCases'], cat_features=[0, 1, 6])
m2.fit(X_train, train_df['Fatalities'], cat_features=[0, 1, 6])
pred_cases = m1.predict(X_test).clip(0)
pred_fat = m2.predict(X_test).clip(0)
submission = pd.DataFrame({
    'ForecastId': test_df['ForecastId'],
    'ConfirmedCases': pred_cases,
    'Fatalities': pred_fat,
})
