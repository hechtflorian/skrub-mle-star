## Model name
LightGBM (LGBMClassifier)

## Example Python code
```python
import pandas as pd
from lightgbm import LGBMClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

for df in [train, test]:
    df[['Deck','CabinNum','Side']] = df['Cabin'].fillna('X/0/X').str.split('/', expand=True)
    df['GroupSize'] = df['PassengerId'].str.split('_').str[0].map(train['PassengerId'].str.split('_').str[0].value_counts())
    df['TotalSpending'] = df[['RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']].fillna(0).sum(axis=1)
    df['HasSpending'] = (df['TotalSpending'] > 0).astype(int)

train = pd.get_dummies(train.drop(columns=['Name','Cabin']), dummy_na=True)
test = pd.get_dummies(test.drop(columns=['Name','Cabin']), dummy_na=True)
train, test = train.align(test, join='left', axis=1, fill_value=0)

X = train.drop(columns=['Transported'])
y = train['Transported'].astype(int)

model = LGBMClassifier(
    n_estimators=2000,
    learning_rate=0.03,
    num_leaves=31,
    subsample=0.8,
    colsample_bytree=0.8,
    random_state=42,
    n_jobs=-1,
    verbose=-1
)
model.fit(X, y)
pred = model.predict(test).astype(bool)
sub = pd.DataFrame({'PassengerId': pd.read_csv('test.csv')['PassengerId'], 'Transported': pred})
sub.to_csv('submission.csv', index=False)
```
