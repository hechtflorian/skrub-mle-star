## Model name
XGBoost Regressor with ordinal-encoded region and temporal features

## Example Python code
import pandas as pd
import numpy as np
from sklearn.preprocessing import OrdinalEncoder
from xgboost import XGBRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

for df in (train, test):
    df['Province_State'] = df['Province_State'].fillna('')
    df['Region'] = df['Country_Region'] + '_' + df['Province_State']
    df['Date'] = pd.to_datetime(df['Date'])
    df['day'] = (df['Date'] - train['Date'].min()).dt.days
    df['month'] = df['Date'].dt.month
    df['dayofweek'] = df['Date'].dt.dayofweek

enc = OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1)
train['region_id'] = enc.fit_transform(train[['Region']])
test['region_id'] = enc.transform(test[['Region']])

features = ['region_id', 'day', 'month', 'dayofweek']
X_train = train[features]
X_test = test[features]

y1 = np.log1p(train['ConfirmedCases'])
y2 = np.log1p(train['Fatalities'])

m1 = XGBRegressor(
    n_estimators=400, max_depth=6, learning_rate=0.05,
    subsample=0.9, colsample_bytree=0.9, objective='reg:squarederror', random_state=42
)
m2 = XGBRegressor(
    n_estimators=400, max_depth=6, learning_rate=0.05,
    subsample=0.9, colsample_bytree=0.9, objective='reg:squarederror', random_state=42
)

m1.fit(X_train, y1)
m2.fit(X_train, y2)

sub = pd.DataFrame({
    'ForecastId': test['ForecastId'],
    'ConfirmedCases': np.expm1(m1.predict(X_test)).clip(0),
    'Fatalities': np.expm1(m2.predict(X_test)).clip(0)
})
sub.to_csv('submission.csv', index=False)
