## Model name
CatBoostRegressor

## Example Python code
import pandas as pd
from catboost import CatBoostRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

X = train.drop(columns=['id', 'yield'])
y = train['yield']
X_test = test.drop(columns=['id'])

model = CatBoostRegressor(
    iterations=2000,
    learning_rate=0.03,
    depth=6,
    loss_function='MAE',
    eval_metric='MAE',
    random_seed=42,
    verbose=200
)

model.fit(X, y)
pred = model.predict(X_test)

sub = pd.DataFrame({'id': test['id'], 'yield': pred})
sub.to_csv('submission_catboost.csv', index=False)
