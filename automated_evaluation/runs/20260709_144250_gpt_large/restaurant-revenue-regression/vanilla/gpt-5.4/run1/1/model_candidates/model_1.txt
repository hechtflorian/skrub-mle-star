## Model name
CatBoostRegressor

## Example Python code
import pandas as pd
from catboost import CatBoostRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

# date features
for df in [train, test]:
    df['Open Date'] = pd.to_datetime(df['Open Date'])
    df['open_year'] = df['Open Date'].dt.year
    df['open_month'] = df['Open Date'].dt.month
    df['open_day'] = df['Open Date'].dt.day
    df['open_weekday'] = df['Open Date'].dt.weekday
    df.drop(columns=['Open Date'], inplace=True)

X = train.drop(columns=['revenue'])
y = train['revenue']
X_test = test.copy()

cat_cols = ['City', 'City Group', 'Type']
cat_idx = [X.columns.get_loc(c) for c in cat_cols]

model = CatBoostRegressor(
    iterations=3000,
    learning_rate=0.03,
    depth=6,
    loss_function='RMSE',
    eval_metric='RMSE',
    random_seed=42,
    verbose=200
)

model.fit(X, y, cat_features=cat_idx)
pred = model.predict(X_test)

sub = pd.DataFrame({'Id': test['Id'], 'Prediction': pred})
sub.to_csv('submission_catboost.csv', index=False)
