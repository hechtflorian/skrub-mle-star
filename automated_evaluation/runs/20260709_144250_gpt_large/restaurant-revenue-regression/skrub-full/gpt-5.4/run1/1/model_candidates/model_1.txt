## Model name
CatBoostRegressor

## Example Python code
import pandas as pd
from catboost import CatBoostRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

for df in (train, test):
    d = pd.to_datetime(df['Open Date'], format='%m/%d/%Y')
    df['restaurant_age_days'] = (pd.Timestamp('2015-01-01') - d).dt.days
    df.drop(columns=['Open Date'], inplace=True)

cat_cols = ['City', 'City Group', 'Type']
feature_cols = [c for c in train.columns if c not in ['Id', 'revenue']]

model = CatBoostRegressor(
    loss_function='RMSE',
    depth=6,
    learning_rate=0.03,
    iterations=700,
    random_seed=42,
    verbose=0
)
model.fit(train[feature_cols], train['revenue'], cat_features=cat_cols)

pred = model.predict(test[feature_cols])
pd.DataFrame({'Id': test['Id'], 'Prediction': pred}).to_csv('submission.csv', index=False)
