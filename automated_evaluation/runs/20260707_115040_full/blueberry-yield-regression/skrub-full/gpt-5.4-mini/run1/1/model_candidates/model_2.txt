## Model name
XGBoost Regressor

## Example Python code
import pandas as pd
from xgboost import XGBRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')
X = train.drop(columns=['id', 'yield'])
y = train['yield']

model = XGBRegressor(
    n_estimators=3000,
    learning_rate=0.03,
    max_depth=6,
    subsample=0.8,
    colsample_bytree=0.8,
    objective='reg:absoluteerror',
    tree_method='hist',
    random_state=42,
    n_jobs=-1,
    verbosity=0
)
model.fit(X, y)
pred = model.predict(test.drop(columns=['id']))
sub = pd.DataFrame({'id': test['id'], 'yield': pred})
sub.to_csv('submission.csv', index=False)
