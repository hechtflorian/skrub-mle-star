## Model name
LightGBM Regressor

## Example Python code
import pandas as pd
from sklearn.model_selection import KFold
from lightgbm import LGBMRegressor

df = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

for c in ['Open Date']:
    df[c] = pd.to_datetime(df[c])
    test[c] = pd.to_datetime(test[c])
    df[c + '_year'] = df[c].dt.year; test[c + '_year'] = test[c].dt.year
    df[c + '_month'] = df[c].dt.month; test[c + '_month'] = test[c].dt.month
    df[c + '_day'] = df[c].dt.day; test[c + '_day'] = test[c].dt.day

df = pd.get_dummies(df.drop(columns=['revenue']), columns=['City', 'City Group', 'Type'])
test = pd.get_dummies(test, columns=['City', 'City Group', 'Type'])
X = df.align(test, join='left', axis=1, fill_value=0)[0]
X_test = df.align(test, join='left', axis=1, fill_value=0)[1]
y = pd.read_csv('train.csv')['revenue']

model = LGBMRegressor(n_estimators=2000, learning_rate=0.03, num_leaves=31, subsample=0.8, colsample_bytree=0.8, random_state=42)
model.fit(X, y)
pred = model.predict(X_test)
pd.DataFrame({'Id': test['Id'], 'Prediction': pred}).to_csv('submission.csv', index=False)
