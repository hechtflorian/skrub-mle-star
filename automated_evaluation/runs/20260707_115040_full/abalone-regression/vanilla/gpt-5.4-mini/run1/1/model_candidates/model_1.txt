## Model name
CatBoostRegressor

## Example Python code
import pandas as pd
import numpy as np
from catboost import CatBoostRegressor, Pool
from sklearn.model_selection import KFold

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')
X = pd.get_dummies(train.drop(columns=['Rings']))
y = np.log1p(train['Rings'])
X_test = pd.get_dummies(test)
X, X_test = X.align(X_test, join='left', axis=1, fill_value=0)

kf = KFold(n_splits=5, shuffle=True, random_state=42)
oof = np.zeros(len(X))
pred = np.zeros(len(X_test))

for tr_idx, va_idx in kf.split(X):
    model = CatBoostRegressor(
        loss_function='RMSE',
        iterations=3000,
        learning_rate=0.03,
        depth=8,
        verbose=False,
        random_seed=42
    )
    model.fit(X.iloc[tr_idx], y.iloc[tr_idx], eval_set=(X.iloc[va_idx], y.iloc[va_idx]), use_best_model=True)
    oof[va_idx] = model.predict(X.iloc[va_idx])
    pred += model.predict(X_test) / kf.n_splits

submission = pd.DataFrame({'id': test['id'], 'Rings': np.expm1(pred).clip(0)})
submission.to_csv('submission.csv', index=False)
