## Model name
CatBoostClassifier

## Example Python code
import pandas as pd
from catboost import CatBoostClassifier

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

y = train['Personality'].map({'Introvert': 0, 'Extrovert': 1})
X = train.drop(columns=['Personality'])
X_test = test.copy()

cat_cols = [c for c in X.columns if X[c].dtype == 'object']

model = CatBoostClassifier(
    iterations=1500,
    depth=6,
    learning_rate=0.03,
    loss_function='Logloss',
    eval_metric='Accuracy',
    verbose=200,
    random_seed=42
)
model.fit(X, y, cat_features=cat_cols)

pred = model.predict(X_test).astype(int).ravel()
sub = pd.DataFrame({'id': test['id'], 'Personality': pd.Series(pred).map({0:'Introvert', 1:'Extrovert'})})
sub.to_csv('submission.csv', index=False)
