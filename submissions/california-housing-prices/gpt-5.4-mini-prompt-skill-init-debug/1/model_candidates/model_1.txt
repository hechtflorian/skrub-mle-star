## Model name
HistGradientBoostingRegressor + skrub DataOps tabular learner

## Example Python code
import pandas as pd
import skrub

train = pd.read_csv("train.csv")
test = pd.read_csv("test.csv")

X = skrub.X(train.drop(columns=["median_house_value"]))
y = skrub.y(train["median_house_value"])

# Strong default for small/medium tabular regression, works well on housing data
learner = skrub.tabular_learner(task="regression")

pred = X.skb.apply(learner, y=y)
search = pred.skb.make_randomized_search(scoring="neg_root_mean_squared_error", n_iter=1, random_state=0, fitted=True)

best = search.best_learner_
submission = pd.DataFrame({"median_house_value": best.predict(test)})
submission.to_csv("submission.csv", index=False)
