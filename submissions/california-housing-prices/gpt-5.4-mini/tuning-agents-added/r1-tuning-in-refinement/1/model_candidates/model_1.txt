## Model name
CatBoostRegressor

## Example Python code
import pandas as pd
from catboost import CatBoostRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

target = 'median_house_value'
X = train.drop(columns=[target])
y = train[target]

model = CatBoostRegressor(
    iterations=2000,
    learning_rate=0.03,
    depth=8,
    loss_function='RMSE',
    random_seed=42,
    verbose=0
)
model.fit(X, y)
preds = model.predict(test)
pd.DataFrame({target: preds}).to_csv('submission.csv', index=False)
