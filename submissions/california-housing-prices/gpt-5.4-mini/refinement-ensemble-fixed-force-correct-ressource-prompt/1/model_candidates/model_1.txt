## Model name
TableVectorizer + HistGradientBoostingRegressor (strong baseline)

## Example Python code
import pandas as pd
import skrub
from sklearn.ensemble import HistGradientBoostingRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

target_col = 'median_house_value'
data = skrub.var('data', train)
X = data.drop(columns=target_col, errors='ignore').skb.mark_as_X()
y = data[target_col].skb.mark_as_y()

vectorizer = skrub.TableVectorizer(
    high_cardinality=skrub.choose_from(
        {
            'minhash': skrub.MinHashEncoder(n_components=skrub.choose_int(8, 32, name='n_components')),
            'gap': skrub.GapEncoder(n_components=skrub.choose_int(8, 32, name='n_components_gap')),
        },
        name='high_card_encoder',
    )
)

model = HistGradientBoostingRegressor(
    learning_rate=skrub.choose_float(0.02, 0.2, log=True, name='learning_rate'),
    max_depth=skrub.choose_int(3, 10, name='max_depth'),
    min_samples_leaf=skrub.choose_int(10, 80, name='min_samples_leaf'),
    random_state=42,
)

pred = X.skb.apply(vectorizer).skb.apply(model, y=y)
learner = pred.skb.make_learner(fitted=True)
train_pred = learner.predict({'data': train})
val_rmse = ((train[target_col] - train_pred) ** 2).mean() ** 0.5
print(f'Final Validation Performance: {val_rmse}')

test_pred = learner.predict({'data': test})
submission = pd.DataFrame({target_col: test_pred})
submission.to_csv('submission.csv', index=False)
