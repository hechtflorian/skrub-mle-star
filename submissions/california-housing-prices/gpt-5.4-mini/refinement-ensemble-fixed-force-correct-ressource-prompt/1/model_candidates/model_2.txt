## Model name
TableVectorizer + CatBoostRegressor (often wins on mixed tabular data)

## Example Python code
import pandas as pd
import skrub
from catboost import CatBoostRegressor

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

target_col = 'median_house_value'
X = skrub.X(train.drop(columns=target_col))
y = skrub.y(train[target_col])

vectorizer = skrub.TableVectorizer(
    high_cardinality=skrub.choose_from(
        {
            'minhash': skrub.MinHashEncoder(n_components=16),
            'gap': skrub.GapEncoder(n_components=16),
        },
        name='high_card_encoder',
    )
)

model = CatBoostRegressor(
    depth=skrub.choose_int(6, 10, name='depth'),
    learning_rate=skrub.choose_float(0.02, 0.2, log=True, name='learning_rate'),
    iterations=skrub.choose_int(500, 2000, name='iterations'),
    loss_function='RMSE',
    verbose=False,
    random_seed=42,
)

pred = X.skb.apply(vectorizer).skb.apply(model, y=y)
learner = pred.skb.make_learner(fitted=True)
train_pred = learner.predict({'data': train})
val_rmse = ((train[target_col] - train_pred) ** 2).mean() ** 0.5
print(f'Final Validation Performance: {val_rmse}')

test_pred = learner.predict({'data': test})
pd.DataFrame({target_col: test_pred}).to_csv('submission.csv', index=False)
