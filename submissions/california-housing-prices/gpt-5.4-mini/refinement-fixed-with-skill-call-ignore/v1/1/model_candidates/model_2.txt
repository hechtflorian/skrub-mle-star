## Model name
RandomForestRegressor + TableVectorizer

## Example Python code
import pandas as pd
import skrub
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_squared_error

train_df = pd.read_csv('train.csv')
test_df = pd.read_csv('test.csv')
target_col = 'median_house_value'

X = skrub.X(train_df.drop(columns=target_col))
y = skrub.y(train_df[target_col])

vec = skrub.TableVectorizer(
    high_cardinality=skrub.choose_from(
        {'gap': skrub.GapEncoder(), 'minhash': skrub.MinHashEncoder(n_components=8)},
        name='high_card_encoder'
    )
)

pred = X.skb.apply(vec).skb.apply(
    RandomForestRegressor(
        n_estimators=skrub.choose_int(300, 800, name='n_estimators'),
        max_depth=skrub.choose_int(10, 30, name='max_depth'),
        min_samples_leaf=skrub.choose_int(1, 4, name='min_samples_leaf'),
        random_state=0,
        n_jobs=-1,
    ),
    y=y,
)
learner = pred.skb.make_learner(fitted=True)
train_pred = learner.predict({'data': train_df.drop(columns=target_col)})
rmse = mean_squared_error(train_df[target_col], train_pred) ** 0.5
print(f'Final Validation Performance: {rmse}')

test_pred = learner.predict({'data': test_df})
pd.DataFrame({target_col: test_pred}).to_csv('submission.csv', index=False)
