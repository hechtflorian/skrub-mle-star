## Model name
Skrub DataOps pipeline with MinHashEncoder + HistGradientBoostingRegressor

## Example Python code
import pandas as pd
import skrub
from sklearn.ensemble import HistGradientBoostingRegressor
from sklearn.metrics import mean_squared_error

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')
target = 'median_house_value'

X = skrub.X(train.drop(columns=[target]))
y = skrub.y(train[target])

pipe = X.skb.apply(
    skrub.TableVectorizer(
        high_cardinality=skrub.MinHashEncoder(n_components=20),
        low_cardinality=skrub.ToCategorical(),
    )
).skb.apply(
    HistGradientBoostingRegressor(random_state=42),
    y=y,
)

learner = pipe.skb.make_learner(fitted=True)
preds = learner.predict({'data': test})

# simple holdout-style check using the last 20% of train
cut = int(len(train) * 0.8)
learner2 = X.iloc[:cut].skb.apply(
    skrub.TableVectorizer(
        high_cardinality=skrub.MinHashEncoder(n_components=20),
        low_cardinality=skrub.ToCategorical(),
    )
).skb.apply(
    HistGradientBoostingRegressor(random_state=42),
    y=y.iloc[:cut],
).skb.make_learner(fitted=True)
valid_pred = learner2.predict({'data': train.iloc[cut:].drop(columns=[target])})
rmse = mean_squared_error(train[target].iloc[cut:], valid_pred) ** 0.5
print(f'Final Validation Performance: {rmse}')

pd.DataFrame({'median_house_value': preds}).to_csv('submission.csv', index=False)
