## Model name
HistGradientBoostingRegressor + TableVectorizer

## Example Python code
import pandas as pd
import skrub
from sklearn.ensemble import HistGradientBoostingRegressor
from sklearn.metrics import mean_squared_error

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

X = skrub.X(train.drop(columns='median_house_value'))
y = skrub.y(train['median_house_value'])

pipe = X.skb.apply(skrub.TableVectorizer()).skb.apply(
    HistGradientBoostingRegressor(random_state=0), y=y
)

pred = pipe.skb.make_learner(fitted=True)
val_pred = pred.predict({'data': train.drop(columns='median_house_value')})
rmse = mean_squared_error(train['median_house_value'], val_pred) ** 0.5
print(f'Final Validation Performance: {rmse}')

test_pred = pred.predict({'data': test})
submission = pd.DataFrame({'median_house_value': test_pred})
submission.to_csv('submission.csv', index=False)
