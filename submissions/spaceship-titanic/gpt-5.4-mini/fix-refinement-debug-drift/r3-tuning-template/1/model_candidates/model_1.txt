## Model name
CatBoostClassifier

## Example Python code
import pandas as pd
from catboost import CatBoostClassifier

# basic preprocessing
train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

def prep(df):
    df = df.copy()
    df[['Deck','Num','Side']] = df['Cabin'].str.split('/', expand=True)
    df['Group'] = df['PassengerId'].str.split('_').str[0]
    for c in ['CryoSleep','VIP','HomePlanet','Destination','Deck','Side']:
        df[c] = df[c].astype('str').fillna('missing')
    num_cols = ['Age','RoomService','FoodCourt','ShoppingMall','Spa','VRDeck']
    df[num_cols] = df[num_cols].fillna(df[num_cols].median(numeric_only=True))
    return df.drop(columns=['Name','Cabin'])

X = prep(train).drop(columns=['Transported'])
y = train['Transported'].astype(int)
X_test = prep(test)

cat_cols = [c for c in X.columns if X[c].dtype == 'object']
model = CatBoostClassifier(
    loss_function='Logloss',
    eval_metric='Accuracy',
    iterations=800,
    depth=6,
    learning_rate=0.03,
    random_seed=42,
    verbose=0
)
model.fit(X, y, cat_features=cat_cols)
preds = model.predict(X_test).astype(bool).ravel()
sub = pd.DataFrame({'PassengerId': test['PassengerId'], 'Transported': preds})
sub.to_csv('submission.csv', index=False)
