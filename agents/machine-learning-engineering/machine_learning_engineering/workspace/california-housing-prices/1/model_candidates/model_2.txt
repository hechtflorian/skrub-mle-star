## Model name
RandomForestRegressor + TableVectorizer

## Example Python code
import pandas as pd
import skrub
from sklearn.ensemble import RandomForestRegressor

train_df = pd.read_csv('train.csv')
test_df = pd.read_csv('test.csv')
target = 'median_house_value'

X = skrub.X(train_df.drop(columns=[target]))
y = skrub.y(train_df[target])

model = RandomForestRegressor(
    n_estimators=600,
    max_depth=None,
    min_samples_leaf=2,
    random_state=0,
    n_jobs=-1,
)

pred = X.skb.apply(skrub.TableVectorizer()).skb.apply(model, y=y)
learner = pred.skb.make_learner(fitted=True)

test_pred = learner.predict({'data': test_df})
submission = pd.DataFrame({target: test_pred})
submission.to_csv('submission.csv', index=False)
