## Model name
LightGBMClassifier

## Example Python code
import pandas as pd
from lightgbm import LGBMClassifier

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

for df in [train, test]:
    df[['Deck','Num','Side']] = df['Cabin'].fillna('X/0/X').str.split('/', expand=True)
    df['Group'] = df['PassengerId'].astype(str).str.split('_').str[0]
    df['FamilySize'] = df['Group'].map(df['Group'].value_counts())
    df['NameLen'] = df['Name'].fillna('').str.len()
    df.drop(['Cabin','Name'], axis=1, inplace=True)

full = pd.concat([train.drop('Transported', axis=1), test], axis=0)
full = pd.get_dummies(full, dummy_na=True)
X = full.iloc[:len(train)]
test_X = full.iloc[len(train):]
y = train['Transported'].astype(int)

X = X.fillna(-1)
test_X = test_X.fillna(-1)

model = LGBMClassifier(n_estimators=1200, learning_rate=0.03, num_leaves=31, subsample=0.8, colsample_bytree=0.8)
model.fit(X, y)
preds = model.predict(test_X)
submission = pd.DataFrame({'PassengerId': test['PassengerId'], 'Transported': preds.astype(bool)})
submission.to_csv('submission.csv', index=False)
