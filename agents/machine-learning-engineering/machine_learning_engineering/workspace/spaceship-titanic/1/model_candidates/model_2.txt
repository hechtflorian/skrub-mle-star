## Model name
LightGBM with simple feature engineering

## Example Python code
import pandas as pd
import numpy as np
from lightgbm import LGBMClassifier
from sklearn.model_selection import train_test_split

train = pd.read_csv('train.csv')
test = pd.read_csv('test.csv')

def fe(df):
    df = df.copy()
    cabin = df['Cabin'].fillna('X/X/X').str.split('/', expand=True)
    df['Deck'] = cabin[0]; df['Num'] = pd.to_numeric(cabin[1], errors='coerce')
    df['Side'] = cabin[2]
    df['GroupSize'] = df.groupby('Group')['PassengerId'].transform('count') if 'Group' in df.columns else 1
    return df

for d in [train, test]:
    d[['HomePlanet','CryoSleep','Cabin','Destination','VIP']] = d[['HomePlanet','CryoSleep','Cabin','Destination','VIP']].astype('object')
    d['Group'] = d['PassengerId'].str.split('_').str[0]
    d = d
train = fe(train); test = fe(test)
y = train['Transported'].astype(int)
X = train.drop(columns=['Transported'])
X = X.fillna(-1); test = test.fillna(-1)
cat = X.select_dtypes(include='object').columns.tolist()
for c in cat:
    X[c] = X[c].astype('category')
    test[c] = test[c].astype('category')
model = LGBMClassifier(n_estimators=1200, learning_rate=0.03, num_leaves=31)
model.fit(X, y)
pred = model.predict(test)
sub = pd.DataFrame({'PassengerId': test['PassengerId'], 'Transported': pred.astype(bool)})
sub.to_csv('submission.csv', index=False)
